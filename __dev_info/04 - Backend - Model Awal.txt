# core/tenants/middleware.py:
from django.http import Http404
from core.tenants.models import Tenant

class TenantResolutionMiddleware:
    """
    Resolve tenant from request header and attach to request.
    """

    HEADER_NAME = "HTTP_X_TENANT_ID"

    def __init__(self, get_response):
        self.get_response = get_response


    def __call__(self, request):
        tenant_id = request.META.get(self.HEADER_NAME)

        if not tenant_id:
            raise Http404("Tenant not specified")        
            
        try:
            tenant = Tenant.objects.get(
                id=tenant_id,
                is_active=True
            )
        except Tenant.DoesNotExist:
            raise Http404("Invalid tenant")
        
        request.tenant = tenant

        if request.user.is_authenticated:
            if request.user.tenant_id != tenant.id:
                raise Http404("User does not belong to tenant")
            
        return self.get_response(request)

# core/roles/models.py:
class Role(models.Model):
    tenant = models.ForeignKey(
        Tenant,
        on_delete=models.CASCADE,
        related_name="roles"
    )

    name = models.CharField(max_length=100)
    description = models.TextField(blank=True)

    users = models.ManyToManyField(
        "core_users.User",
        through="core_roles.UserRole",
        related_name="roles"
    )

    class Meta:
        db_table = "core_roles"
        unique_together = ("tenant", "name")

    
    def __str__(self):
        return self.name
    

class UserRole(models.Model):
    user = models.ForeignKey(
        "core_users.User",
        on_delete=models.CASCADE,
        related_name="user_roles"
    )
    role = models.ForeignKey(
        "core_roles.Role",
        on_delete=models.CASCADE,
        related_name="role_users"
    )

    assigned_at = models.DateTimeField(auto_now_add=True)

    
    class Meta:
        db_table = "core_user_roles"
        unique_together = ("user", "role")

    
    def __str__(self):
        return f"{self.user} → {self.role}"

# core/permissions/middleware.py
class PermissionGuardMiddleware:
    """
    Enforce permission if view declares required_permission.
    """

    def __init__(self, get_response):
        self.get_response = get_response

    
    def __call__(self, request):
        return self.get_response(request)
    

    def process_view(self, request, view_func, view_args, view_kwargs):
        required_permission = getattr(
            view_func,
            "required_permission",
            None
        )

        if not required_permission:
            return None
        
        if not PermissionService.can_access(
            request.user,
            required_permission
        ):
            raise PermissionDenied(
                f"Permission '{required_permission}' required"
            )
        
        return None


# core/permissions/service.py
from core.permissions.models import Permission
from core.users.models import User


class PermissionService:
    """
    Centralized permission evaluation.
    DO NOT push business logic here.
    """

    @staticmethod
    def user_has_permission(
        user: User,
        permission_code: str
    ) -> bool:
        """
        Check whether a user has a given permission via roles.
        """

        if not user or not user.is_authenticated:
            return False
        
        return Permission.objects.filter(
            code=permission_code,
            permission_roles__role__role_user__user=user,
            permission_roles__role__tenant=user.tenant
        ).exists()
    
    
    @staticmethod
    def user_has_any_permission(
        user: User,
        permission_codes: list[str]
    ) -> bool:
        if not user or not user.is_authenticated:
            return False
        
        return Permission.objects.filter(
            code__in=permission_codes,
            permission_roles__role__role_users__user=user,
            permission_roles__role__tenant=user.tenant
        ).exists()
    

    @staticmethod
    def can_access(
        user: User,
        permission_code: str
    ) -> bool:
        if not user or not user.is_authenticated:
            return False
        
        # Platform override
        if user.is_superuser:
            return True
        
        return PermissionService.user_has_permission(
            user,
            permission_code
        )
